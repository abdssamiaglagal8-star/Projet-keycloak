\begin{tcolorbox}[colback=s2mlight,colframe=s2mblue,boxrule=0.5pt,left=4pt,right=4pt,top=2pt,bottom=2pt,arc=2pt]
  \footnotesize\textbf{Objectif.} Activer sur le \textbf{Browser Flow} : (1) \textbf{S2M Turnstile} (CAPTCHA anti-bots),
  (2) Username/Password, (3) \textbf{S2M Email OTP} (code 6 chiffres par e-mail).\\
  \textbf{Pr\'er\'equis :} JAR \texttt{keycloak-security-extension} d\'eploy\'e + th\`eme S2M
  (\texttt{captcha-turnstile.ftl}, \texttt{email-otp.ftl}) + red\'emarrage Keycloak.
\end{tcolorbox}

\begin{stepbox}{\'Etape 1 --- Choisir le Realm}
\begin{enumerate}\footnotesize
  \item Ouvrir la console admin Keycloak : \texttt{https://<host>/admin}.
  \item En haut \`a \textbf{gauche}, cliquer sur le s\'electeur de realm.
  \item S\'electionner le realm de l'application (ex. \texttt{s2m} / \texttt{itsp}).
\end{enumerate}
\screenshot{Admin $\rightarrow$ s\'electeur Realm (en-t\^ete gauche)}
\end{stepbox}

\begin{stepbox}{\'Etape 2 --- Menu Authentication}
\begin{enumerate}\footnotesize
  \item Menu gauche : \textbf{Authentication}.
  \item Onglet \textbf{Flows}.
  \item Rep\'erer le flux natif \texttt{browser} (connexion web).
\end{enumerate}
\screenshot{Authentication $\rightarrow$ Flows $\rightarrow$ liste (browser, \ldots)}
\end{stepbox}

\begin{stepbox}{\'Etape 3 --- Cr\'eer un flux personnalis\'e (copie)}
\begin{enumerate}\footnotesize
  \item Sur la ligne \texttt{browser} $\rightarrow$ menu \textbf{\ldots} / Action $\rightarrow$ \textbf{Duplicate} (ou Copy).
  \item Nommer le flux : \texttt{browser-s2m-mfa}.
  \item Valider. \textit{(Ne jamais modifier le flux browser d'origine.)}
\end{enumerate}
\screenshot{Duplicate browser $\rightarrow$ nom \guillemotleft{}browser-s2m-mfa\guillemotright{}}
\end{stepbox}

\begin{stepbox}{\'Etape 4 --- Ajouter le CAPTCHA (S2M Turnstile)}
\begin{enumerate}\footnotesize
  \item Ouvrir le flux \texttt{browser-s2m-mfa}.
  \item \textbf{Add step} / Add execution.
  \item Choisir : \textbf{\guillemotleft{}S2M Cloudflare Turnstile CAPTCHA\guillemotright{}} (ID : \texttt{s2m-turnstile}).
  \item Placer \textbf{en premier} dans le flux. Exigence : \textbf{REQUIRED}.
\end{enumerate}
\screenshot{Add step $\rightarrow$ S2M Cloudflare Turnstile CAPTCHA $\rightarrow$ REQUIRED}
\end{stepbox}

\begin{stepbox}{\'Etape 5 --- Configurer les cl\'es Turnstile (Cloudflare)}
\begin{enumerate}\footnotesize
  \item Sur la ligne Turnstile $\rightarrow$ ic\^one \textbf{Config} (engrenage).
  \item Alias ex. : \texttt{turnstile-prod}. Renseigner :
\end{enumerate}
{\footnotesize
\begin{center}
\begin{tabular}{@{}>{\bfseries}p{3.6cm} p{5.4cm} p{5.2cm}@{}}
\toprule
Param\`etre & R\^ole & Exemple \\
\midrule
\texttt{siteKey} & Cl\'e publique (widget) & \texttt{0x4AAAAA\ldots} \\
\texttt{secretKey} & Cl\'e secr\`ete serveur & \texttt{0x4AAAAA\ldots} \textit{(jamais en Git)} \\
\texttt{verifyUrl} & API Cloudflare & \texttt{https://challenges.cloudflare.com/turnstile/v0/siteverify} \\
\texttt{timeoutSeconds} & Timeout HTTP & \texttt{5} \\
\bottomrule
\end{tabular}
\end{center}}
\textcolor{s2morange}{\footnotesize\textbf{Important :}} secret uniquement dans l'admin Keycloak.
\screenshot{Config Turnstile --- siteKey + secretKey}
\end{stepbox}

  \item Position : \textbf{apr\`es} Username Password (user d\'ej\`a identifi\'e). Exigence : \textbf{REQUIRED}.
  \item Config :

\begin{stepbox}{\'Etape 7 --- Lier le flux (Browser Flow binding)}
\begin{enumerate}\footnotesize
  \item \textbf{Authentication} $\rightarrow$ onglet \textbf{Bindings} (Flow bindings).
  \item Champ \textbf{Browser Flow} $\rightarrow$ s\'electionner \texttt{browser-s2m-mfa} $\rightarrow$ Save.
\end{enumerate}
Toute connexion web passe d\'esormais par Captcha + OTP.
\screenshot{Authentication $\rightarrow$ Bindings $\rightarrow$ Browser Flow = browser-s2m-mfa}
\end{stepbox}

\begin{stepbox}{\'Etape 8 --- Th\`eme de login (\'ecrans utilisateur)}
\footnotesize Fichiers du th\`eme S2M : \texttt{captcha-turnstile.ftl}, \texttt{email-otp.ftl},
templates e-mail \texttt{email/html/email-otp.ftl} \& \texttt{email/text/email-otp.ftl}.
\begin{enumerate}\footnotesize
  \item \textbf{Realm settings} $\rightarrow$ \textbf{Themes}.
  \item \textbf{Login theme} et \textbf{Email theme} = th\`eme S2M (ex. \texttt{s2m-theme-v2}) $\rightarrow$ Save.
\end{enumerate}
\screenshot{Realm settings $\rightarrow$ Themes $\rightarrow$ Login / Email = s2m-theme-v2}
\end{stepbox}

\begin{stepbox}{\'Etape 9 --- SMTP (obligatoire pour l'OTP e-mail)}
\begin{enumerate}\footnotesize
  \item \textbf{Realm settings} $\rightarrow$ \textbf{Email} : h\^ote, port, from, auth.
  \item Bouton \textbf{Test connection} --- v\'erifier r\'eception d'un mail de test.
\end{enumerate}
Sans SMTP valide, l'\'etape OTP \'echoue \`a l'envoi du code.
\screenshot{Realm settings $\rightarrow$ Email $\rightarrow$ SMTP + Test connection}
\end{stepbox}

\section*{Parcours utilisateur (v\'erification)}
{\footnotesize
\begin{enumerate}
  \item Ouvrir l'URL de login de l'application.
  \item \textbf{\'Ecran 1 --- CAPTCHA :} widget Turnstile $\rightarrow$ valider $\rightarrow$ Continuer.
  \item \textbf{\'Ecran 2 --- Login :} username + password.
  \item \textbf{\'Ecran 3 --- OTP :} code 6 chiffres re\c{c}u par e-mail (5 min / 5 essais max).
  \item Acc\`es si code correct.
\end{enumerate}}
